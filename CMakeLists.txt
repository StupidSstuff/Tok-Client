cmake_minimum_required(VERSION 3.24)
project(TokClient VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows only")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build x64 only")
endif()
add_library(TokClient SHARED src/dllmain.cpp src/version.cpp)
target_compile_features(TokClient PRIVATE cxx_std_20)
target_include_directories(TokClient PRIVATE include)
target_compile_definitions(TokClient PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_link_libraries(TokClient PRIVATE kernel32)
set_target_properties(TokClient PROPERTIES OUTPUT_NAME "TokClient")
